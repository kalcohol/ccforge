# Mock only compiler results and build-target creation; load the real test gates.
function(check_cxx_source_compiles source result_var)
    if(result_var STREQUAL "FORGE_SUBMDSPAN_HAS_MDSPAN_HEADER")
        set(result ON)
    elseif(result_var STREQUAL "FORGE_SUBMDSPAN_PARTIAL_HAS_PORTABLE_OPERATIONS")
        if(NOT GATE_CASE MATCHES "^partial_")
            message(FATAL_ERROR "COMPLETE/backport portable tests must not be capability-gated")
        endif()
        set(result "${_partial_available}")
    elseif(result_var STREQUAL "FORGE_SUBMDSPAN_ACCEPTS_PADDED_CONVERSIONS")
        set(result ON)
        if(GATE_CASE STREQUAL "positive_control_failure")
            set(result OFF)
        endif()
    elseif(result_var MATCHES "^FORGE_SUBMDSPAN_REJECTS_")
        set(result OFF)
        if(GATE_CASE STREQUAL "negative_control_success")
            set(result ON)
        endif()
    else()
        message(FATAL_ERROR "Unexpected compiler probe: ${result_var}")
    endif()
    set(${result_var} "${result}" CACHE INTERNAL "Synthetic compiler result" FORCE)
endfunction()

# Match the shared probe helpers' cache and PARENT_SCOPE contract.
function(forge_detect_cxx_source_compiles result_var source_file)
    unset(${result_var} CACHE)
    unset(${result_var})
    file(READ "${source_file}" source)
    check_cxx_source_compiles("${source}" ${result_var})
    set(${result_var} "${${result_var}}" PARENT_SCOPE)
endfunction()

function(forge_expect_cxx_source_compiles result_var source_file should_compile failure_message)
    forge_detect_cxx_source_compiles(${result_var} "${source_file}")
    if((should_compile AND NOT ${result_var}) OR
            (NOT should_compile AND ${result_var}))
        message(FATAL_ERROR "${failure_message}")
    endif()
endfunction()

function(add_executable target)
    add_custom_target(${target})
    set(excluded OFF)
    if("EXCLUDE_FROM_ALL" IN_LIST ARGN)
        set(excluded ON)
    endif()
    set_property(TARGET ${target} PROPERTY FORGE_FIXTURE_EXCLUDED "${excluded}")
endfunction()

function(target_link_libraries target)
    if(NOT TARGET ${target})
        message(FATAL_ERROR "Unknown link target: ${target}")
    endif()
endfunction()

function(target_compile_definitions target)
    set_property(TARGET ${target} PROPERTY FORGE_FIXTURE_DEFINITIONS "${ARGN}")
endfunction()

function(target_include_directories target)
    if(NOT TARGET ${target})
        message(FATAL_ERROR "Unknown include target: ${target}")
    endif()
endfunction()

function(forge_add_runtime_test target source name)
    add_executable(${target} "${source}")
    add_test(NAME ${name} COMMAND ${target})
endfunction()

function(forge_add_compile_probe target source name)
    add_executable(${target} EXCLUDE_FROM_ALL "${source}")
    add_test(NAME ${name} COMMAND ${CMAKE_COMMAND} --build . --target ${target})
endfunction()

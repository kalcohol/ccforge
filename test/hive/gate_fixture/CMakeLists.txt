cmake_minimum_required(VERSION 3.20)
project(ForgeHiveTestGates LANGUAGES NONE)
if(NOT DEFINED FORGE_ROOT_DIR OR NOT DEFINED GATE_CASE)
    message(FATAL_ERROR "FORGE_ROOT_DIR and GATE_CASE are required")
endif()
enable_testing()
set(FORGE_TEST_TIMEOUT_SECONDS 30)
set(FORGE_HIVE_FULL OFF)
set(FORGE_HIVE_PARTIAL OFF)
set(FORGE_FORCE_HIVE_BACKPORT OFF)
if(GATE_CASE MATCHES "complete$")
    set(FORGE_HIVE_FULL ON)
    set(FORGE_HIVE_PARTIAL ON)
elseif(GATE_CASE MATCHES "partial$")
    set(FORGE_HIVE_PARTIAL ON)
elseif(NOT GATE_CASE STREQUAL "backport")
    message(FATAL_ERROR "Unknown gate case: ${GATE_CASE}")
endif()
if(GATE_CASE MATCHES "^forced_")
    set(FORGE_FORCE_HIVE_BACKPORT ON)
endif()

# Replace build targets only; exercise the real registration and gate logic.
function(add_executable target)
    add_custom_target(${target})
    set(excluded FALSE)
    if("EXCLUDE_FROM_ALL" IN_LIST ARGN)
        set(excluded TRUE)
    endif()
    set_property(TARGET ${target} PROPERTY FORGE_FIXTURE_EXCLUDED ${excluded})
endfunction()
# Preserve names and properties without evaluating executable-only command
# expressions such as TARGET_FILE against the mocked utility targets.
function(add_test)
    if(NOT ARGV0 STREQUAL "NAME" OR NOT ARGV2 STREQUAL "COMMAND")
        message(FATAL_ERROR "Unexpected test registration syntax: ${ARGV}")
    endif()
    if(FIXTURE_DROP_PORTABLE AND ARGV1 STREQUAL "hive_portable_runtime")
        return()
    endif()
    _add_test(NAME ${ARGV1} COMMAND "${CMAKE_COMMAND}" -E true)
endfunction()
function(add_library target)
    add_custom_target(${target})
endfunction()
function(target_link_libraries target)
endfunction()
function(target_include_directories target)
endfunction()
function(target_compile_definitions target)
    if(target STREQUAL "hive_portable")
        message(FATAL_ERROR "Portable tests must not receive implementation definitions")
    endif()
endfunction()
include("${FORGE_ROOT_DIR}/test/hive/RegisterTests.cmake")
get_directory_property(actual_tests DIRECTORY "${FORGE_ROOT_DIR}/test/hive" TESTS)
set(expected_tests hive_portable_runtime)
foreach(probe IN ITEMS fancy_api accept_from_range reject_incompatible_from_range
        accept_construct_only_range accept_assign_range reject_nonassignable_assign_range
        accept_iterator_allocator reject_non_allocator accept_legacy_input reject_non_input_iterator)
    list(APPEND expected_tests hive_compile_${probe})
endforeach()
if(GATE_CASE STREQUAL "backport" OR FORGE_FORCE_HIVE_BACKPORT)
    list(APPEND expected_tests hive_runtime hive_multi_tu hive_model_positive hive_model_mutation hive_bounded_sort)
    foreach(probe IN ITEMS include api constexpr_allocator version_then_include)
        list(APPEND expected_tests hive_compile_${probe})
    endforeach()
endif()
list(SORT actual_tests)
list(SORT expected_tests)
if(NOT actual_tests STREQUAL expected_tests)
    message(FATAL_ERROR "Gate ${GATE_CASE}: expected ${expected_tests}; got ${actual_tests}")
endif()
if(NOT TARGET hive_portable)
    message(FATAL_ERROR "Portable target missing")
endif()
get_target_property(excluded hive_portable FORGE_FIXTURE_EXCLUDED)
if(excluded)
    message(FATAL_ERROR "Portable target must be in the default build")
endif()
foreach(target IN ITEMS hive_runtime hive_multi_tu hive_compile_api hive_model_normal hive_bounded_sort)
    if(GATE_CASE STREQUAL "backport" OR FORGE_FORCE_HIVE_BACKPORT)
        if(NOT TARGET ${target})
            message(FATAL_ERROR "Implementation target missing: ${target}")
        endif()
    elseif(TARGET ${target})
        message(FATAL_ERROR "Native gate leaked implementation target: ${target}")
    endif()
endforeach()

cmake_minimum_required(VERSION 3.20)
project(ForgeConfigurationConsumer LANGUAGES CXX)

set(_empty_equivalent_inputs
    CMAKE_REQUIRED_FLAGS CMAKE_REQUIRED_DEFINITIONS CMAKE_REQUIRED_INCLUDES
    CMAKE_REQUIRED_LIBRARIES CMAKE_REQUIRED_LINK_OPTIONS
    CMAKE_REQUIRED_LINK_DIRECTORIES)
if(CASE MATCHES "^same_(empty|undefined)$")
    foreach(_input IN LISTS _empty_equivalent_inputs)
        # Hide any parent cache entry as well as its ordinary variable.
        unset(${_input} CACHE)
        if(CASE STREQUAL "same_empty")
            unset(${_input})
        else()
            set(${_input} "")
        endif()
    endforeach()
endif()
if(CASE STREQUAL "different_extensions")
    unset(CMAKE_CXX_EXTENSIONS)
    unset(CMAKE_CXX_EXTENSIONS CACHE)
endif()
if(CASE STREQUAL "unset" OR CASE STREQUAL "source_default")
    unset(CMAKE_CXX_STANDARD)
    unset(CMAKE_CXX_STANDARD CACHE)
endif()
if(CASE STREQUAL "different_platform")
    list(APPEND CMAKE_TRY_COMPILE_PLATFORM_VARIABLES FORGE_FIXTURE_PLATFORM)
    set(FORGE_FIXTURE_PLATFORM first)
endif()
if(CASE STREQUAL "c_language")
    enable_language(C)
endif()
if(CASE MATCHES "^(different|same)_header$")
    set(_fixture_header "${CMAKE_CURRENT_BINARY_DIR}/content-input/hive")
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/content-input")
    file(WRITE "${_fixture_header}" "// First header contents.\n")
    # A search root tracked by the shared resolver, without shadowing the real
    # standard library during compilation of the probe itself.
    list(PREPEND CMAKE_INCLUDE_PATH "${CMAKE_CURRENT_BINARY_DIR}/content-input")
elseif(CASE MATCHES "^(different|same)_toolchain$")
    set(_fixture_toolchain "${CMAKE_CURRENT_BINARY_DIR}/content-toolchain.cmake")
    file(WRITE "${_fixture_toolchain}" "# First toolchain contents.\n")
    # Preserve the parent toolchain when nested try_compile loads this wrapper.
    if(CMAKE_TOOLCHAIN_FILE)
        if(IS_ABSOLUTE "${CMAKE_TOOLCHAIN_FILE}")
            set(_parent_toolchain "${CMAKE_TOOLCHAIN_FILE}")
        elseif(EXISTS "${CMAKE_BINARY_DIR}/${CMAKE_TOOLCHAIN_FILE}")
            set(_parent_toolchain "${CMAKE_BINARY_DIR}/${CMAKE_TOOLCHAIN_FILE}")
        else()
            set(_parent_toolchain "${CMAKE_SOURCE_DIR}/${CMAKE_TOOLCHAIN_FILE}")
        endif()
        file(APPEND "${_fixture_toolchain}" "include([==[${_parent_toolchain}]==])\n")
    endif()
    set(CMAKE_TOOLCHAIN_FILE "${_fixture_toolchain}")
endif()
if(MODE STREQUAL "subdirectory")
    add_subdirectory("${FORGE_SOURCE}" forge-source)
elseif(MODE STREQUAL "fetchcontent")
    include(FetchContent)
    FetchContent_Declare(Forge SOURCE_DIR "${FORGE_SOURCE}")
    FetchContent_MakeAvailable(Forge)
elseif(MODE STREQUAL "package")
    find_package(CCForge CONFIG REQUIRED)
else()
    include("${FORGE_SOURCE}/forge.cmake")
endif()

if(CASE STREQUAL "source_default")
    set(_expected_standard 23)
else()
    set(_expected_standard "${CMAKE_CXX_STANDARD}")
endif()
if(NOT FORGE_PROBE_CXX_STANDARD STREQUAL _expected_standard)
    # add_subdirectory keeps ordinary output variables in its own scope.
    get_property(FORGE_PROBE_CXX_STANDARD GLOBAL PROPERTY _FORGE_PROBE_CXX_STANDARD)
endif()
if(NOT FORGE_PROBE_CXX_STANDARD STREQUAL _expected_standard)
    message(FATAL_ERROR "Forge did not record the probe standard")
endif()

if(CASE MATCHES "^same_(empty|undefined)$")
    foreach(_input IN LISTS _empty_equivalent_inputs)
        if(CASE STREQUAL "same_empty")
            set(${_input} "")
        else()
            unset(${_input})
        endif()
    endforeach()
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE STREQUAL "different_extensions")
    set(CMAKE_CXX_EXTENSIONS OFF)
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE STREQUAL "same" AND MODE STREQUAL "package")
    find_package(CCForge CONFIG REQUIRED)
elseif(CASE STREQUAL "same")
    # Exercise a second source entry point after add_subdirectory/FetchContent.
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE MATCHES "^(different|same)_(header|toolchain)$")
    if(CASE STREQUAL "different_header")
        file(APPEND "${_fixture_header}" "// Changed at the same path.\n")
    elseif(CASE STREQUAL "different_toolchain")
        file(APPEND "${_fixture_toolchain}" "# Changed at the same path.\n")
    endif()
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE STREQUAL "different")
    set(FORGE_FORCE_SIMD_BACKPORT ON)
    set(FORGE_ENABLE_FORGE_RUNTIME OFF)
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE STREQUAL "different_standard")
    set(CMAKE_CXX_STANDARD 26)
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE STREQUAL "different_deployment")
    set(CMAKE_OSX_DEPLOYMENT_TARGET "${CMAKE_OSX_DEPLOYMENT_TARGET}-changed")
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE STREQUAL "different_platform")
    set(FORGE_FIXTURE_PLATFORM second)
    include("${FORGE_SOURCE}/forge.cmake")
elseif(CASE STREQUAL "different_environment")
    set(ENV{CPLUS_INCLUDE_PATH} "$ENV{CPLUS_INCLUDE_PATH}/forge-fixture-changed")
    include("${FORGE_SOURCE}/forge.cmake")
endif()

if(CASE STREQUAL "private")
    add_library(implementation STATIC private_library.cpp)
    target_link_libraries(implementation PRIVATE forge::forge)
    add_executable(consumer private_main.cpp)
    set_property(TARGET consumer PROPERTY CXX_STANDARD 20)
    target_link_libraries(consumer PRIVATE implementation)
else()
    add_executable(consumer main.cpp)
    target_link_libraries(consumer PRIVATE forge::forge)
endif()
if(CASE STREQUAL "mismatch")
    set_property(TARGET consumer PROPERTY CXX_STANDARD 26)
elseif(CASE STREQUAL "features")
    target_compile_features(consumer PRIVATE cxx_std_26)
elseif(CASE STREQUAL "transitive")
    add_library(language INTERFACE)
    target_compile_features(language INTERFACE cxx_std_26)
    target_link_libraries(consumer PRIVATE language)
elseif(CASE STREQUAL "promotion")
    # The raw property differs; the transitive feature supplies the probe mode.
    set_property(TARGET consumer PROPERTY CXX_STANDARD 23)
    add_library(language INTERFACE)
    target_compile_features(language INTERFACE cxx_std_26)
    target_link_libraries(consumer PRIVATE language)
endif()

if(CASE STREQUAL "c_language")
    target_sources(consumer PRIVATE c_guard.c)
    # Compile C directly with the generated guard pre-included, independently
    # of Forge usage requirements or the generator's COMPILE_LANGUAGE support.
    add_library(c_guard_direct OBJECT c_guard.c)
    set(_guard "${CMAKE_BINARY_DIR}/CMakeFiles/CCForgeLanguageGuard.h")
    if(MSVC OR CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
        target_compile_options(c_guard_direct PRIVATE "/FI${_guard}")
    else()
        target_compile_options(c_guard_direct PRIVATE "SHELL:-include \"${_guard}\"")
    endif()
    add_dependencies(consumer c_guard_direct)
endif()
